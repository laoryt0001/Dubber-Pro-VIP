📖 Client SDK Integration

Drop licenseClient.js into your Electron app and activate with:

const { LicenseClient } = require('./licenseClient');

const client = new LicenseClient({
  serverUrl: 'http://localhost:3000'
});

// Activate license when user enters key
async function activateApp(userKey) {
  const result = await client.activate(userKey);
  if (result.valid) {
    console.log('License valid for plan:', result.plan);
    console.log('Features enabled:', result.features);
  } else {
    console.error('License error:', result.error);
  }
}
